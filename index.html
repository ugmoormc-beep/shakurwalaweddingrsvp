function ensureHeaders_(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Timestamp", "Full Name", "Phone Number", "Email (Optional)", "Attendance", "Number of Guests", "Wishes", "CreatedAtMs"]);
  }
}

function parseIncomingPayload_(e) {
  var params = (e && e.parameter) ? e.parameter : {};
  if (params.payload) {
    try { return JSON.parse(params.payload); } catch (err) {}
  }
  if (e && e.postData && e.postData.contents) {
    try { return JSON.parse(e.postData.contents); } catch (err2) {}
  }
  if (params.fullName) {
    return {
      id: params.id || ("row_" + new Date().getTime()),
      timestamp: params.timestamp || new Date().toLocaleString(),
      fullName: params.fullName || "",
      phone: params.phone || "",
      email: params.email || "",
      attendance: params.attendance || "Joyfully Attending",
      guests: Number(params.guests || 0),
      wishes: params.wishes || "",
      createdAtMs: Number(params.createdAtMs || new Date().getTime()),
      action: params.action || "insert"
    };
  }
  return null;
}

function insertRow_(sheet, data) {
  ensureHeaders_(sheet);
  if (!data || !data.fullName) return { status: "empty" };
  var id = String(data.id || ("row_" + new Date().getTime()));
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var values = sheet.getRange(1, 1, lastRow, 1).getValues();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][0]) === id) {
        return { status: "duplicate", id: id };
      }
    }
  }
  sheet.appendRow([
    id,
    data.timestamp || new Date().toLocaleString(),
    data.fullName || "",
    data.phone || "",
    data.email || "—",
    data.attendance || "Joyfully Attending",
    Number(data.guests || 0),
    data.wishes || "",
    Number(data.createdAtMs || new Date().getTime())
  ]);
  return { status: "success", id: id };
}

function deleteRowById_(sheet, targetId) {
  var values = sheet.getDataRange().getValues();
  for (var i = values.length - 1; i >= 1; i--) {
    if (String(values[i][0]) === String(targetId)) {
      sheet.deleteRow(i + 1);
      return { status: "deleted", id: targetId };
    }
  }
  return { status: "not_found", id: targetId };
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = parseIncomingPayload_(e);
    if (!data) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No payload" })).setMimeType(ContentService.MimeType.JSON);
    }
    var res = (data.action === "delete" && data.id) ? deleteRowById_(sheet, data.id) : insertRow_(sheet, data);
    return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  ensureHeaders_(sheet);
  var params = (e && e.parameter) ? e.parameter : {};
  if (params.action === "insert" || params.fullName || params.payload) {
    var lock = LockService.getScriptLock();
    lock.tryLock(10000);
    try {
      var payloadObj = parseIncomingPayload_(e);
      var writeRes = insertRow_(sheet, payloadObj);
      return outputJsonOrJsonp_(writeRes, params.callback);
    } finally {
      lock.releaseLock();
    }
  }
  if (params.action === "delete" && params.id) {
    var delRes = deleteRowById_(sheet, params.id);
    return outputJsonOrJsonp_(delRes, params.callback);
  }
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[2]) continue;
    result.push({
      id: String(r[0] || ("sheet_" + i)),
      timestamp: String(r[1] || ""),
      fullName: String(r[2] || ""),
      phone: String(r[3] || ""),
      email: String(r[4] === "—" ? "" : (r[4] || "")),
      attendance: String(r[5] || "Joyfully Attending"),
      guests: Number(r[6] || 0),
      wishes: String(r[7] || ""),
      createdAtMs: Number(r[8] || (new Date().getTime() - (rows.length - i) * 1000))
    });
  }
  return outputJsonOrJsonp_({ status: "ok", items: result }, params.callback);
}

function outputJsonOrJsonp_(obj, callback) {
  var json = JSON.stringify(obj);
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + json + ");").setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
